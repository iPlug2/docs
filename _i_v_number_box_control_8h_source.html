49 void GetAccessibleNodes(IAccessibleNodeList& nodes) override
50 {
51 if (mAccessibleRole == EAccessibleRole::None)
52 return;
53
54 nodes.emplace_back();
55 IAccessibleNode& node = nodes.back();
56 InitAccessibleNode(node, mAccessibleRole);
57 node.hasNumericValue = true;
58 node.min = mMinValue;
59 node.max = mMaxValue;
60 node.value = mRealValue;
61 node.step = mLargeIncrement;
62
63 if (mTextReadout)
64 node.valueText.Set(mTextReadout->GetStr());
65 }
66
67 bool OnAccessibleAction(EAccessibleAction action, int nodeIdx = 0, double value = 1.) override
68 {
69 if (IsDisabled())
70 return false;
71
72 const int nSteps = std::max(1, static_cast<int>(std::round(value)));
73
74 switch (action)
75 {
76 case EAccessibleAction::Increment: mRealValue += nSteps * mLargeIncrement; break;
77 case EAccessibleAction::Decrement: mRealValue -= nSteps * mLargeIncrement; break;
78 case EAccessibleAction::SetValue: mRealValue = value; break;
79 default: return false;
80 }
81
82 if (GetParam())
83 GetDelegate()->BeginInformHostOfParamChangeFromUI(GetParamIdx());
84
85 OnValueChanged();
86
87 if (GetParam())
88 GetDelegate()->EndInformHostOfParamChangeFromUI(GetParamIdx());
89
90 return true;
91 }

99 mRealValue = GetParam()->GetDefault();
100 }
101 }

118 }

121 void OnResize() override
122 {
123 MakeRects(mRECT, false);
124
125 IRECT sections = mWidgetBounds;
126 sections.Pad(-1.f);
127
128 if (mTextReadout)
129 {
130 mTextReadout->SetTargetAndDrawRECTs(sections.ReduceFromLeft(sections.W() * (mButtons ? 0.75f : 1.f)));
131
132 if (mButtons)
133 {
134 mIncButton->SetTargetAndDrawRECTs(sections.FracRectVertical(0.5f, true).GetPadded(-2.f, 0.f, 0.f, -1.f));
135 mDecButton->SetTargetAndDrawRECTs(sections.FracRectVertical(0.5f, false).GetPadded(-2.f, -1.f, 0.f, 0.f));
136 }
137
138 SetTargetRECT(mTextReadout->GetRECT());
139 }
140 }

142 void OnAttached() override
143 {
144 IRECT sections = mWidgetBounds;
145 sections.Pad(-1.f);
146
147 AddChildControl(mTextReadout = new IVLabelControl(sections.ReduceFromLeft(sections.W() * (mButtons ? 0.75f : 1.f)), "0", mStyle.WithDrawFrame(true)));
148
149 mTextReadout->SetStrFmt(32, mFmtStr.Get(), mRealValue);
150 mTextReadout->SetAccessibleRole(EAccessibleRole::None); // the number box itself is exposed as a slider
151
152 if (mButtons)
153 {
154 AddChildControl(mIncButton = new IVButtonControl(sections.FracRectVertical(0.5f, true).GetPadded(-2.f, 0.f, 0.f, -1.f), SplashClickActionFunc, "+", mStyle))->SetAnimationEndActionFunction(mIncrementFunc);
155 AddChildControl(mDecButton = new IVButtonControl(sections.FracRectVertical(0.5f, false).GetPadded(-2.f, -1.f, 0.f, 0.f), SplashClickActionFunc, "-", mStyle))->SetAnimationEndActionFunction(mDecrementFunc);
156 mIncButton->SetAccessibleRole(EAccessibleRole::None);
157 mDecButton->SetAccessibleRole(EAccessibleRole::None);
158 }
159 }

185 void OnMouseDrag(float x, float y, float dX, float dY, const IMouseMod &mod) override
186 {

188 mRealValue -= (double(dY) * gearing);

196 }

202 }
203
204 void OnMouseWheel(float x, float y, const IMouseMod& mod, float d) override
205 {
206 double gearing = IsFineControl(mod, true) ? mSmallIncrement : mLargeIncrement;
207 double inc = (d > 0.f ? 1. : -1.) * gearing;
208 mRealValue += inc;
209 OnValueChanged();
210 }

259 void OnValueChanged(bool preventAction = false)
260 {
261 mRealValue = Clip(mRealValue, mMinValue, mMaxValue);
262
263 mTextReadout->SetStrFmt(32, mFmtStr.Get(), mRealValue);
264
265 if (!preventAction && GetParam())
266 SetValue(GetParam()->ToNormalized(mRealValue));
267
268 SetDirty(!preventAction);
269 }
