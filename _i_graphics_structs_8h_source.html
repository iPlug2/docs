318 void GetHSLA(float& h, float& s, float& l, float& a) const
319 {
320 const float fR = R / 255.f;
321 const float fG = G / 255.f;
322 const float fB = B / 255.f;
323 a = A / 255.f;
324
325 const float fMin = std::min(fR, std::min(fG, fB));
326 const float fMax = std::max(fR, std::max(fG, fB));
327 const float fDiff = fMax - fMin;
328 const float fSum = fMax + fMin;
329
330 l = 50.f * fSum;
331
332 if (fMin == fMax) { s = 0.f; h = 0.f; l /= 100.f; return; }
333 else if (l < 50.f) { s = 100.f * fDiff / fSum; }
334 else { s = 100.f * fDiff / (2.f - fDiff); }
335
336 if (fMax == fR) { h = 60.f * (fG - fB) / fDiff; }
337 if (fMax == fG) { h = 60.f * (fB - fR) / fDiff + 120.f; }
338 if (fMax == fB) { h = 60.f * (fR - fG) / fDiff + 240.f; }
339
340 if (h < 0.f) { h = h + 360.f; }
341
342 h /= 360.f;
343 s /= 100.f;
344 l /= 100.f;
345 }

459 static IColor FromHSLA(float h, float s, float l, float a = 1.f)
460 {
461 auto hue = [](float h, float m1, float m2) {
462 if (h < 0) h += 1;
463 if (h > 1) h -= 1;
464 if (h < 1.0f / 6.0f)
465 return m1 + (m2 - m1) * h * 6.0f;
466 else if (h < 3.0f / 6.0f)
467 return m2;
468 else if (h < 4.0f / 6.0f)
469 return m1 + (m2 - m1) * (2.0f / 3.0f - h) * 6.0f;
470 return m1;
471 };
472
473 IColor col;
474 h = std::fmod(h, 1.0f);
475 if (h < 0.0f) h += 1.0f;
476 s = Clip(s, 0.0f, 1.0f);
477 l = Clip(l, 0.0f, 1.0f);
478 float m2 = l <= 0.5f ? (l * (1 + s)) : (l + s - l * s);
479 float m1 = 2 * l - m2;
480 col.R = static_cast<int>(Clip(hue(h + 1.0f / 3.0f, m1, m2), 0.0f, 1.0f) * 255.f);
481 col.G = static_cast<int>(Clip(hue(h, m1, m2), 0.0f, 1.0f) * 255.f);
482 col.B = static_cast<int>(Clip(hue(h - 1.0f / 3.0f, m1, m2), 0.0f, 1.0f) * 255.f);
483 col.A = static_cast<int>(a * 255.f);
484 return col;
485 }

1913 {
1914 for (int j = i + 1; j < Size(); j++)
1915 {
1916 if (Get(i).Contains(Get(j)))
1917 {

1919 j--;
1920 }
1921 else if (Get(j).Contains(Get(i)))

1932 Set(i, Shrink(Get(i), intersection));
1933 else if (Get(j).Mergeable(intersection))
1934 Set(j, Shrink(Get(j), intersection));

1946 {
